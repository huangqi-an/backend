### 2. `config/env.ts` —— 强类型环境变量

```ts
import "dotenv/config";

function required(name: string): string {
	const v = process.env[name];
	if (!v) throw new Error(`Missing env var: ${name}`);
	return v;
}

export const env = {
	nodeEnv: required("NODE_ENV") as "development" | "production" | "test",
	port: parseInt(required("PORT"), 10),
	sessionSecret: required("SESSION_SECRET"),
	sessionName: required("SESSION_NAME"),
	redis: {
		host: required("REDIS_HOST"),
		port: parseInt(required("REDIS_PORT"), 10),
		password: process.env.REDIS_PASSWORD || undefined,
	},
	sessionTtl: parseInt(process.env.SESSION_TTL ?? "86400", 10),
	isProd: required("NODE_ENV") === "production",
} as const;
```

### 3. `lib/redis.ts` —— Redis 单例 client

`redis` v6 用 `createClient`，必须 `await client.connect()` 后才能使用。

```ts
import { createClient, type RedisClientType } from "redis";
import { env } from "../config/env.js";

let client: RedisClientType | null = null;

export async function getRedisClient(): Promise<RedisClientType> {
	if (client) return client;

	client = createClient({
		socket: { host: env.redis.host, port: env.redis.port },
		password: env.redis.password,
	}) as RedisClientType;

	client.on("error", (err) => {
		console.error("[Redis] client error", err);
	});

	await client.connect();
	return client;
}
```

> 注意：ESM + `nodenext` 下相对导入必须带 `.js` 扩展名（即使源文件是 `.ts`），由 `rewriteRelativeImportExtensions: true` 配合 Node 的要求。

### 5. `middleware/session.ts` —— express-session + Redis Store

```ts
import session from "express-session";
import { RedisStore } from "connect-redis"; // v10 为命名导出
import { env } from "../config/env.js";
import { getRedisClient } from "../lib/redis.js";

export async function buildSessionMiddleware() {
	const redisClient = await getRedisClient();

	return session({
		store: new RedisStore({ client: redisClient, prefix: "sess:" }),
		name: env.sessionName, // cookie 名，默认 connect.sid
		secret: env.sessionSecret,
		resave: false,
		saveUninitialized: false,
		rolling: false, // 每次请求是否刷新过期；按需
		cookie: {
			httpOnly: true,
			sameSite: "lax", // 第一道 CSRF 防线
			secure: env.isProd, // 生产必须 HTTPS
			maxAge: env.sessionTtl * 1000,
		},
		// connect-redis 会自动用 cookie.maxAge 作为 TTL，也可显式 ttl
		// ttl: env.sessionTtl,
	});
}
```

要点：

- `resave: false` + `saveUninitialized: false`：只在 session 真正写入数据时才存 Redis，减少无用键。
- `sameSite: "lax"`：默认阻止跨站非 GET 请求带 cookie，是 CSRF 的第一道防线；token 是第二道。
- 生产部署在 nginx 后面要 `app.set("trust proxy", 1)`，否则 `secure` cookie 在 HTTP 代理下会被丢弃。

### 6. `utils/crypto.util.ts` —— 安全比较 + 随机 token

```ts
import { randomBytes, timingSafeEqual } from "node:crypto";

export function generateCsrfToken(): string {
	return randomBytes(32).toString("base64url");
}

export function safeEqual(a: string, b: string): boolean {
	const ab = Buffer.from(a);
	const bb = Buffer.from(b);
	if (ab.length !== bb.length) return false;
	return timingSafeEqual(ab, bb);
}
```

> 必须用 `timingSafeEqual` 防时序攻击，不要直接 `===` 比对 token。

### 7. `middleware/csrf.ts` —— 自定义 CSRF 中间件（Synchronizer Token 模式）

设计思路：

- 前端先 `GET /csrf/token` 拿一个 token，后端把它存进 session（`session.csrfToken`）并返回。
- 对“不安全方法” `POST/PUT/PATCH/DELETE`，从请求头 `X-CSRF-Token` 或 body `_csrf` 取 token，与 session 里的比对。
- `GET/HEAD/OPTIONS` 不校验（幂等且不应有副作用）。
- token 可重复使用（直到 session 过期/重新生成），对 SPA 友好；登录成功后重新生成防 fixation。

```ts
import type { Request, Response, NextFunction } from "express";
import { generateCsrfToken, safeEqual } from "../utils/crypto.util.js";

const UNSAFE_METHODS = new Set(["POST", "PUT", "PATCH", "DELETE"]);

/** 生成（或复用）token，供 GET /csrf/token 使用 */
export function issueCsrfToken(req: Request, res: Response) {
	if (!req.session.csrfToken) {
		req.session.csrfToken = generateCsrfToken();
	}
	res.json({ csrfToken: req.session.csrfToken });
}

/** 对写操作校验 token */
export function csrfGuard(req: Request, res: Response, next: NextFunction) {
	if (!UNSAFE_METHODS.has(req.method)) {
		return next();
	}

	const sessionToken = req.session.csrfToken;
	const clientToken =
		(req.headers["x-csrf-token"] as string | undefined) ||
		(req.body && typeof req.body._csrf === "string"
			? req.body._csrf
			: undefined);

	if (!sessionToken || !clientToken) {
		return res.status(403).json({ error: "CSRF token missing" });
	}
	if (!safeEqual(clientToken, sessionToken)) {
		return res.status(403).json({ error: "CSRF token invalid" });
	}

	return next();
}
```

可选增强（按需加入）：

- 登出时 `req.session.destroy()`，Redis 会自动删 key。
- 登录成功后调用 `regenerateCsrf(req)`：重新 `generateCsrfToken()` 覆盖旧值并返回新 token。
- 若要严格一次性 token：校验通过后 `delete req.session.csrfToken`，但前端每次写操作前需重新取 token，体验较差。

### 9. `app.ts` —— 组装

中间件顺序很重要：`session` 必须在 CSRF 之前（CSRF 依赖 `req.session`）；`csrfGuard` 要放在“写路由”之前，但**不能**拦截 `GET /csrf/token`。

```ts
import express, { type Express } from "express";
import helmet from "helmet";
import cors from "cors";
import cookieParser from "cookie-parser";
import { env } from "./config/env.js";
import { buildSessionMiddleware } from "./middleware/session.js";
import { csrfGuard } from "./middleware/csrf.js";
import { csrfRouter } from "./routes/csrf.routes.js";

export async function createApp(): Promise<Express> {
	const app = express();

	app.set("trust proxy", 1); // 生产在反向代理后；开发可设 0/false

	app.use(helmet());
	app.use(cors({ origin: true, credentials: true })); // 跨域带 cookie
	app.use(express.json());
	app.use(cookieParser());

	// session 必须在依赖 req.session 的中间件之前
	app.use(await buildSessionMiddleware());

	// 取 token：放 guard 之前
	app.use(csrfRouter); // /csrf/token

	// 对写操作统一拦截
	app.use(csrfGuard);

	// 业务路由（演示）
	app.post("/api/login", (req, res) => {
		req.session.userId = "u1"; // 触发写入 Redis
		res.json({ ok: true });
	});
	app.post("/api/danger", (req, res) => {
		res.json({ ok: true, userId: req.session.userId });
	});

	return app;
}
```

> `cookie-parser` 在纯 session-token 模式下其实非必需（express-session 自己管 cookie），但保留它可解析额外下发的 cookie（如双重提交 cookie 方案）。

## 四、运行流程与时序

1. 前端首次 `GET /csrf/token` → 后端生成 token 存入 session（此时才会往 Redis 写 `sess:<id>`），通过 `Set-Cookie: sid=...` 下发会话 cookie，body 返回 `{ csrfToken }`。
2. 前端把 `csrfToken` 存内存，后续写操作带 `X-CSRF-Token: <token>`，同时浏览器自动带 `sid` cookie。
3. 后端 `csrfGuard` 比对 `req.headers['x-csrf-token']` 与 `req.session.csrfToken`（`timingSafeEqual`）。
4. 登录成功后重新生成 token 返回新值；登出 `req.session.destroy()`，Redis 自动删键。

```
浏览器                            Express                      Redis
  |  GET /csrf/token                |                            |
  |-------------------------------->|  (无 session) 生成 token     |
  |                                 |  SET sess:<id> csrfToken   |
  |                                 |--------------------------->|
  |<--- Set-Cookie: sid=... --------|                            |
  |<--- { csrfToken: "..." } ------ |                            |
  |                                 |                            |
  |  POST /api/danger               |                            |
  |  X-CSRF-Token: ...              |                            |
  |  Cookie: sid=...                |                            |
  |-------------------------------->|  GET sess:<id>             |
  |                                 |--------------------------->|
  |                                 |  timingSafeEqual(客户端,服务端) |
  |<--- { ok: true } ---------------|                            |
```

## 五、关键校验清单

- [ ] 所有 `.ts` 之间的相对 import 带 `.js` 扩展名（`nodenext`）。
- [ ] 类型导入用 `import type`（`verbatimModuleSyntax`）。
- [ ] 不用 `enum` / `namespace` / 参数属性（`erasableSyntaxOnly`），改 `as const` 联合类型。
- [ ] `express-session` 在 `helmet` / `cors` / `json` / `cookieParser` 之后、CSRF 之前挂载。
- [ ] Redis `client.connect()` 在构建 app 时 `await`，不要懒加载到第一个请求。
- [ ] 生产：`secure: true` cookie + `trust proxy` + HTTPS；`SESSION_SECRET` 足够长。
- [ ] token 比对用 `timingSafeEqual`，不能用 `===`。
- [ ] `GET /csrf/token` 在 `csrfGuard` 之前注册，否则永远拿不到 token。

## 六、CSRF 方案选型对比

| 方案                                 | token 存放                         | 优缺点                                                     |
| ------------------------------------ | ---------------------------------- | ---------------------------------------------------------- |
| **Session Synchronizer（本文主推）** | 后端 session + 前端 header         | 安全性高，无额外 cookie；需 session/Redis                  |
| Double Submit Cookie                 | 下发 csrf cookie + 前端回传 header | 无状态友好，但依赖 cookie，需 `cookie-parser`              |
| SameSite Cookie Only                 | 仅靠 `sameSite=strict`             | 最简单，但跨站子请求/旧浏览器有边角问题，建议与 token 并用 |

本项目已装 Redis + session，所以 **Synchronizer Token** 是最贴合、最省心的选择。

## 七、登出实现

### 原理

`req.session.destroy(callback)` 是 express-session 方法，内部调 store 的 `destroy(sessionID, cb)`。connect-redis v10 实现该方法，等价于 Redis `DEL sess:<sessionID>`，所以**不需要**自己写 `redisClient.del(...)`。

**关键坑**：`destroy()` 只删 Redis 里的 session 数据，**不会清除浏览器 cookie**（express-session 官方明确）。登出必须两步：

1. `req.session.destroy()` —— 删 Redis 数据
2. `res.clearCookie(name, options)` —— 清浏览器 cookie

### 实现（routes/auth.router.ts）

```ts
import { Router, type Request, type Response } from "express";
import { env } from "../config/env.ts";

export const authRoutes = Router();

authRoutes.post("/api/login", (req, res) => {
	req.session.userId = "u1";
	res.json({ ok: true });
});

authRoutes.post("/api/logout", (req: Request, res: Response) => {
	// 没有会话：直接清 cookie 并返回
	if (!req.session) {
		res.clearCookie(env.sessionName, {
			httpOnly: true,
			sameSite: "lax",
			secure: env.isProd,
		});
		return res.json({ ok: true, message: "logged out" });
	}
	// 删 Redis 里的 session（connect-redis 自动 DEL sess:<sid>）
	req.session.destroy((err) => {
		if (err) {
			console.error("[logout] destroy error", err);
			return res.status(500).json({ error: "logout failed" });
		}
		// 清浏览器 cookie（path/domain 需与 session cookie 一致）
		res.clearCookie(env.sessionName, {
			httpOnly: true,
			sameSite: "lax",
			secure: env.isProd,
		});
		return res.json({ ok: true, message: "logged out" });
	});
});
```

### 挂载顺序

登出是 POST，需在 csrfGuard 之后（登出也受 CSRF 保护，防借身份触发登出）：

```ts
app.use(await buildSessionMiddleware());
app.use(csrfRouter);     // GET /csrf/token（guard 之前）
app.use(csrfGuard);      // 拦截写操作
app.use(authRoutes);     // POST /api/login、/api/logout
```

### 常见坑

| 坑 | 说明 |
| --- | --- |
| 只 destroy 不 clearCookie | Redis 删了但浏览器 cookie 还在，下次请求带失效 sid |
| clearCookie 的 path/domain 不一致 | 浏览器不删 cookie |
| 登出路由放 csrfGuard 前 | 登出不受 CSRF 保护 |
| destroy 回调里再用 req.session | destroy 后 session 已 unset，会报错 |
| `unset` 选项 | 显式 `destroy()` 不受 `unset` 影响，仍删 store |

### 与 CSRF token 联动

登出删 Redis session 后旧 csrfToken 消失。登录成功后应重新生成 token 防 fixation：

```ts
req.session.userId = user.id;
req.session.csrfToken = generateCsrfToken(); // 覆盖旧值
res.json({ csrfToken: req.session.csrfToken, user });
```

### 实测验证

`docker exec redis redis-cli --scan --pattern 'sess:*'` 观察：logout 前存在当前 session key，logout 后该 key 被删除（`req.session.destroy()` 触发 connect-redis `DEL`）。

### 4. 实测结果

| 请求 | 期望 | 实际 |
| --- | --- | --- |
| GET /csrf/token | 200 + token | ✅ |
| POST /api/danger 无 token | 403 missing | ✅ |
| POST /api/danger 错 token | 403 invalid | ✅ |
| POST /api/danger 对 token | 200 | ✅ |
| POST /api/login 对 token | 200 | ✅ |
| POST /api/logout 对 token | 200 + 删 Redis key | ✅ |
| logout 后旧 token 再 POST | 403 | ✅ |

### 5. 关于"两个 Redis key"

若用 `GET /csrf/token` 做就绪检测，每次无 cookie 的请求都会创建一个孤儿 session（`issueCsrfToken` 写 csrfToken 触发保存，但 cookie 未保存导致再无法访问）。孤儿随 TTL 自动清理，非 bug。用 `GET /health` 做就绪检测即可避免：

```ts
app.get("/health", (_req, res) => res.json({ ok: true }));
```

## 九、Docker 环境编排

> 编排范围：compose 只起 Redis；应用默认宿主机 `node index.ts` 跑。多阶段 Dockerfile 为应用容器化的可选备用资产。compose 放根目录 `backend/`，Dockerfile 放 `nodejs/`（依赖 hoisted 在该层，构建上下文需包含 `package.json` + `pnpm-lock.yaml` + `express/`）。

### 根级 docker-compose.yml

```yaml
services:
  redis:
    image: redis:7-alpine
    container_name: backend-redis
    restart: unless-stopped
    command: ["redis-server", "--requirepass", "${REDIS_PASSWORD}", "--appendonly", "yes"]
    environment:
      REDISCLI_AUTH: ${REDIS_PASSWORD}
    ports: ["${REDIS_PORT:-6379}:6379"]
    volumes: [redis-data:/data]
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 5
    networks: [backend-net]

  # app（可选，默认不启用）：
  #   build: { context: ./nodejs }
  #   env_file: .env
  #   environment:
  #     REDIS_HOST: redis
  #     NODE_ENV: production
  #   ports: ["${PORT:-3000}:3000"]
  #   depends_on: { redis: { condition: service_healthy } }
  #   networks: [backend-net]

volumes:
  redis-data:
networks:
  backend-net:
```

要点：

- `${REDIS_PASSWORD}` 等由 `backend/.env`（compose 同目录自动读）插值。
- `REDISCLI_AUTH` 让 healthcheck 的 `redis-cli ping` 自动认证，不暴露密码。
- `appendonly yes` + `redis-data` 卷 = 重启不丢数据。
- 注释的 app 服务里 `REDIS_HOST: redis` 是关键差异：容器化后连服务名，不再连 `127.0.0.1`。

### nodejs/Dockerfile（多阶段，可选）

```dockerfile
# syntax=docker/dockerfile:1
FROM node:24-alpine AS deps
WORKDIR /app
RUN corepack enable
COPY package.json pnpm-lock.yaml ./
RUN --mount=type=cache,id=pnpm,target=/root/.local/share/pnpm/store \
    pnpm install --frozen-lockfile
COPY . .

FROM node:24-alpine AS runtime
WORKDIR /app
ENV NODE_ENV=production
COPY --from=deps /app /app
WORKDIR "/app/express/Session + Cookie + CSRF Token"
EXPOSE 3000
CMD ["node", "index.ts"]
```

## 十、环境变量与两套 .env

### 三个层次

| 层次 | 来源 | 作用 |
| --- | --- | --- |
| compose `${VAR}` 插值 | `backend/.env`（compose 同目录自动读） | 决定 redis 容器密码、端口 |
| 容器内进程 `process.env` | `environment:` / `env_file:` 显式注入 | redis 靠 `command --requirepass`；app 靠 `env_file` |
| 本地 `node index.ts` 的 `process.env` | 子项目 `.env`（`dotenv/config` 加载 cwd 的 .env） | app 连 Redis 的配置 |

### 关键区分

- `backend/.env` 仅供 **compose 编排插值**与（容器化 app 时）注入，**不会自动**进容器进程环境。
- 本地宿主机跑 `node index.ts` 读的是**子项目自己的 `.env`**，与 `backend/.env` 独立。
- 因此两边 `REDIS_PASSWORD` 必须**手动保持一致**（均 `123456`），否则 app 连不上 redis。

### 容器化 app 时的差异

容器内 app 连 Redis 用**服务名** `redis`（compose 网络 DNS），不再是 `127.0.0.1`：

```yaml
environment:
  REDIS_HOST: redis   # 而非 127.0.0.1
```

优先级：shell 环境变量 > `.env` 文件（compose 插值时）。
